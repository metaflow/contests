for %loop = 2; %loop > 0; %loop-- {
    read %a[10] subsctracting '0' each time
    read %b[0] # space
    read %b[5] subsctracting '0' each time
    %matches = 5
    for %off = 5; %off > 0; %off-- {
        %x = 0
        %c[5] = 0
        # compare
        if %b[0] %c[0]++
        if %b[1] %c[1]++
        ..
        if %b[5] %c[5]++
        if %a[0] %c[0]--
        ..
        if %a[5] %c[0]--
        if %c[0] %x++
        ..
        if %c[5] %x++
        if %x %matches--
        # shift a left
        %a[0] = 0
        if %a[1] %a[0]++
        %a[1] = 0
        ..
        if %a[9] %a[8]++
        %a[9] = 0
    }
    %r = 49
    if %matches == 0 %r--
    out %r
}
